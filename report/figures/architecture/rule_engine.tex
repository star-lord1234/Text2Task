\node[decision] (cue) at (0,-17) {intent cue\\ matched?};

\node[rule, minimum width=50mm] (matcher) at (56,-55) {Lexical matcher over open tasks\\ \tiny token overlap, owner, recency, anaphora, explicit\\ \tiny ids: MATCHED / AMBIGUOUS\_REFERENT / NO\_REFERENT};
\node[warn, minimum width=34mm] (amb) at (-44,-55) {Clarification\\ \tiny referent chooser; nothing\\ \tiny changes until a person picks};

\draw[flow] (match) -- node[note, above]{several} (amb); \draw[dflow] (matcher) -- (match); \draw[flow] (match) -- node[note, right]{one} (prop);
\draw[flow] (cue) -- node[note, above]{no} (create);
\draw[flow] (create) -- node[note, right]{yes} (crt);
\draw[flow] (prop) -- (rev); \draw[flow] (crt) |- (rev); \draw[flow] (amb) |- ([yshift=2mm]rev.west);
